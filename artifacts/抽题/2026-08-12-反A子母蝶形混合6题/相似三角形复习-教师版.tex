\section*{相似模型混合练习}

\needspace{8\baselineskip}

\begin{problem}[points=10]
  \noindent
  \begin{minipage}[t]{\dimexpr\linewidth-60mm-6mm\relax}
    \vspace{0pt}
    如图，在 $\triangle ABC$ 中，$\angle ABD=\angle ACB$。

已知 $AC=8$，$CD=4$。求 $AB$ 的长。
    
  \end{minipage}\hfill
  \begin{diagramcoltikz}{60mm}{}
  \input{\detokenize{../../题库/2026-07-16-子母型相似/items/Q037/build/diagram/jobs/question_bank-nested-v2-q037-prompt/rendered/prompt.fragment.tex}}
  \end{diagramcoltikz}
  \par\medskip
  \answerarea[54mm]
  \setcounter{solstep}{0}
  \begin{solutionblock}
    \textbf{答案：}$AB=4\sqrt{2}$。\par\medskip
    \solstep{先求小三角形的共线边}{因为点 $D$ 在线段 $AC$ 上，所以 $AD=AC-CD=8-4=4$。}
    \solstep{由等角判相似}{由题设等角以及 $A$ 点的公共角，得 $\triangle ABD\sim\triangle ACB$。}
    \solstep{锁定重复出现的对应边}{对应关系为 $AB\leftrightarrow AC$、$AD\leftrightarrow AB$，因此线段 $AB$ 在比例链中重复出现。}
    \solstep{把共线边转成对应边}{由 $\dfrac{AC}{AB}=\dfrac{AB}{AD}$，得 $AB^2=AC\cdot AD=8\times 4$。}
    \solstep{求值并验算}{边长取正，得 $AB=4\sqrt{2}$；代回可得 $\dfrac{AC}{AB}=\dfrac{AB}{AD}$。}
    
  \end{solutionblock}
\end{problem}

\begin{problem}[points=10]
  \noindent
  \begin{minipage}[t]{\dimexpr\linewidth-60mm-6mm\relax}
    \vspace{0pt}
    如图，$\angle OAC=\angle ODB$。

已知 $AC=\sqrt{6}$，$AO=2\sqrt{3}$，$DB=2$。求 $OD$ 的长。
    
  \end{minipage}\hfill
  \begin{diagramcoltikz}{60mm}{}
  \input{\detokenize{../../题库/2026-07-16-蝶形相似/items/Q006/build/diagram/jobs/question_bank-butterfly-v2-q006-prompt/rendered/prompt.fragment.tex}}
  \end{diagramcoltikz}
  \par\medskip
  \answerarea[54mm]
  \setcounter{solstep}{0}
  \begin{solutionblock}
    \textbf{答案：}$OD=2\sqrt{2}$。\par\medskip
    \solstep{整理已知边长比}{题中给出的两边均为整数或整系数根式，因此可直接写成 $AC:AO=\sqrt{6}:2\sqrt{3}$。}
    \solstep{由等角判相似}{由题设等角和构型自带的另一组等角，得 $\triangle AOC\sim\triangle DOB$。}
    \solstep{按顶点确认对应边}{三组小三角形边与大三角形边依次为：$AO\leftrightarrow OD$，$OC\leftrightarrow OB$，$AC\leftrightarrow DB$。}
    \solstep{保持比例方向一致}{统一保持两组对应边的方向，可列 $\dfrac{AC}{DB}=\dfrac{AO}{OD}$，再代入 $DB=2$。}
    \solstep{求值并验算}{解得 $OD=2\sqrt{2}$；代回后两组对应边的比完全相等。}
    
  \end{solutionblock}
\end{problem}

  \begin{diagramcoltikz}{60mm}{}

  \end{diagramcoltikz}
  \par\medskip
  \answerarea[54mm]
  \setcounter{solstep}{0}
  \begin{solutionblock}
    \textbf{答案：}$CD=\sqrt{6}$。\par\medskip
    \solstep{由等角判相似}{由题设等角以及 $A$ 点的公共角，得 $\triangle ABD\sim\triangle ACB$。}
    \solstep{锁定重复出现的对应边}{对应关系为 $AB\leftrightarrow AC$、$AD\leftrightarrow AB$，因此可用 $AB$ 连接两组对应边。}
    \solstep{由对应边反求共线边}{由 $\dfrac{AC}{AB}=\dfrac{AB}{AD}$，即 $AB^2=AC\cdot AD$，得 $AD=\dfrac{AB^2}{AC}=2\sqrt{6}$。}
    \solstep{处理同一直线上的整段与分段}{点 $D$ 在线段 $AC$ 上，所以 $CD=AC-AD=3\sqrt{6}-2\sqrt{6}$。}
    \solstep{求值并验算}{化简得 $CD=\sqrt{6}$；并且 $AD+CD=AC$。}
    
  \end{solutionblock}
\end{problem}
