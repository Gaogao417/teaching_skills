\section*{初三数学期末练习卷 · 原题第 25 题 · 教师版}

\section*{三、解答题}

\needspace{8\baselineskip}

\begin{problem}[points=14]
  \noindent
  \begin{minipage}[t]{\dimexpr\linewidth-66mm-6mm\relax}
    \vspace{0pt}
    在 $\triangle ABC$ 中，$\angle BAC\ge90^\circ$，点 $D$ 为边 $AC$ 上一点，$\angle ABD=\angle ACB$，$CD=2$。将 $\triangle ABD$ 沿 $BD$ 翻折得到 $\triangle A_1BD$，点 $A_1$ 恰好落在边 $BC$ 的垂直平分线上。
\begin{enumerate}
\item 如图，如果点 $A_1$ 在边 $BC$ 上，求 $\dfrac{AD}{AB}$ 的值；
\item 如图，如果 $AD=CD$，求 $BD$ 的长；
\item 如果 $AB=CD$，求 $\angle BAC$ 的度数。
\end{enumerate}
    
  \end{minipage}\hfill
  \begin{diagramcoltikz}{66mm}{}
  \includegraphics[width=\linewidth]{\detokenize{assets/prompt-01.png}}
  \end{diagramcoltikz}
  \par\medskip
  \setcounter{solstep}{0}
  \begin{solutionblock}
    \textbf{答案：}（1）$\frac{\sqrt3}{3}$；（2）$\sqrt6+\sqrt2$；（3）$96^\circ$。\par\medskip
    \solstep{第（1）问}{由题意，$\triangle ABD\cong\triangle A_1BD$，且 $A_1$ 是 $BC$ 的中点，所以 $S_{\triangle ABD}=S_{\triangle A_1BD}=S_{\triangle A_1CD}$，从而 $\dfrac{S_{\triangle ABD}}{S_{\triangle ACB}}=\dfrac13$。
又因为 $\angle ABD=\angle ACB$，$\angle A=\angle A$，所以 $\triangle ABD\sim\triangle ACB$。
因此 $\dfrac{S_{\triangle ABD}}{S_{\triangle ACB}}=\left(\dfrac{AD}{AB}\right)^2=\dfrac13$，所以 $\dfrac{AD}{AB}=\dfrac{\sqrt3}{3}$。}
    \solstep{第（2）问}{联结 $AA_1,A_1C$。由题意，$BD$ 垂直平分 $AA_1$，交 $AA_1$ 于点 $M$。
所以 $AM=A_1M=\dfrac12AA_1$，$\angle AMD=90^\circ$。
因为 $AD=CD=2$，所以 $\dfrac{AM}{MA_1}=\dfrac{AD}{DC}=1$，从而 $DM\parallel A_1C$，$DM=\dfrac12A_1C$。
所以 $\angle AMD=\angle AA_1C=90^\circ$。
因为 $\angle ABD=\angle ACB$，且 $\angle BAD$ 是公共角，所以 $\triangle ABD\sim\triangle ACB$，从而 $\dfrac{AB}{AC}=\dfrac{AD}{AB}$。
因为 $AD=CD=2$，所以 $AB=2\sqrt2$。
由题意，$\triangle ABD\cong\triangle A_1BD$，所以 $A_1B=AB=2\sqrt2$。
因为点 $A_1$ 在 $BC$ 的垂直平分线上，所以 $A_1B=A_1C=2\sqrt2$，从而 $DM=\sqrt2$。
在 $\mathrm{Rt}\triangle ADM$ 中，$AM=\sqrt{AD^2-DM^2}=\sqrt2$；在 $\mathrm{Rt}\triangle ABM$ 中，$BM=\sqrt{AB^2-AM^2}=\sqrt6$。
所以 $BD=BM+DM=\sqrt6+\sqrt2$。}
    \solstep{第（3）问}{过点 $D$ 作 $DN\parallel AA_1$，交 $A_1C$ 于点 $N$。
因为 $\dfrac{A_1N}{A_1C}=\dfrac{AD}{AC}$，且由 $\triangle ABD\sim\triangle ACB$ 得 $\dfrac{AD}{AB}=\dfrac{AB}{AC}$。
又因为 $AB=CD=2$，所以 $\dfrac{AD}{CD}=\dfrac{CD}{AC}$。点 $D$ 为线段 $AC$ 的黄金分割点，所以 $\dfrac{AD}{CD}=\dfrac{CD}{AC}=\dfrac{\sqrt5-1}{2}$。
因此 $AD=\sqrt5-1$，$A_1D=\sqrt5-1$，$AC=\sqrt5+1$。
又因为 $A_1C=A_1B=AB=2$，所以 $A_1N=3-\sqrt5$，$CN=\sqrt5-1$。
所以 $\dfrac{A_1N}{A_1D}=\dfrac{3-\sqrt5}{\sqrt5-1}=\dfrac{\sqrt5-1}{2}$。
因为 $\dfrac{A_1D}{A_1C}=\dfrac{A_1N}{A_1D}$，且 $\angle NA_1D=\angle DA_1C$，所以 $\triangle A_1ND\sim\triangle A_1DC$。
设 $\angle A_1CD=\angle A_1DN=\alpha$，则 $\angle DA_1C=\angle CDA_1=2\alpha$。
在 $\triangle A_1CD$ 中，$\angle DA_1C+\angle CDA_1+\angle A_1CD=180^\circ$，所以 $\alpha=36^\circ$。
因此 $\angle A_1DN=\angle NDC=36^\circ$。
因为 $DN\parallel AA_1$，所以 $\angle CAA_1=\angle NDC=36^\circ$，从而 $A_1A=A_1C$。
又因为 $AB=A_1B=A_1C$，所以 $AB=A_1B=A_1A$，$\triangle ABA_1$ 是等边三角形，$\angle BAA_1=60^\circ$。
所以 $\angle BAC=\angle BAA_1+\angle CAA_1=60^\circ+36^\circ=96^\circ$。}
