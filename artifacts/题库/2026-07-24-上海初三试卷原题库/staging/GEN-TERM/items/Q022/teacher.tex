  \begin{solutionblock}
    \textbf{答案：}（1）$25^\circ$ 档；（2）$15\sqrt3\text{ cm}$；（3）约 $4.8\text{ cm}$。\par\medskip
    \solstep{第（1）问：比较三档拉伸力}{$F=mg\sin\alpha=50\text{ kg}\times10\text{ N/kg}\times\sin20^\circ\approx500\text{ N}\times0.34=170\text{ N}$；
$F\approx500\text{ N}\times0.42=210\text{ N}$；
$F=500\text{ N}\times0.50=250\text{ N}$。
因为 $170<180<210<220<250$，所以小海选择 $25^\circ$ 档能达到锻炼目的。}
    \solstep{第（2）问：求水平距离}{在 $\mathrm{Rt}\triangle DBC$ 中，$\angle BCD=90^\circ$，$BD=30\text{ cm}$，$\angle ABC=30^\circ$，所以 $BC=BD\cos\angle ABC=30\times\frac{\sqrt3}{2}=15\sqrt3\text{ cm}$。}
    \solstep{第（3）问：求竖直下降距离}{在 $\mathrm{Rt}\triangle DBC$ 中，$DC=DB\sin\angle ABC=30\sin30^\circ=15\text{ cm}$。
作 $D_1H\perp BM$，垂足为 $H$。由题意，$D_1B=DB=30\text{ cm}$，$\angle A_1BC_1=20^\circ$。
在 $\mathrm{Rt}\triangle D_1BH$ 中，$D_1H=D_1B\sin\angle A_1BC_1=30\sin20^\circ\approx30\times0.34=10.2\text{ cm}$。
所以点 $D$ 在竖直方向下降约 $15-10.2=4.8\text{ cm}$。}
