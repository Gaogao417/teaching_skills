\begin{problem}[points=12]
  \noindent
  \begin{minipage}[t]{\dimexpr\linewidth-60mm-6mm\relax}
    \vspace{0pt}
    如图，在 $\triangle ABC$ 中，点 $D$ 在线段 $AC$ 上，$BD\perp BC$，且
$\angle ABD=\angle ACB$。若 $AD:CD=9:7$，求 $\cos C$。

    
  \end{minipage}\hfill
  \begin{diagramcoltikz}{60mm}{第 2 题图}
  \input{\detokenize{build/diagram/jobs/practice-q2-prompt/rendered/prompt.fragment.tex}}
  \end{diagramcoltikz}
  \par\medskip
  \answerarea[45mm]
  \setcounter{solstep}{0}
  \begin{solutionblock}
    \textbf{答案：}$\cos C=\dfrac45$\par\medskip
    \solstep{模型一：求相似输出边比}{设 $AD=9x$、$CD=7x$，则 $AC=16x$。由 $AB^2=AD\cdot AC$ 得 $AB=12x$，所以 $BD:BC=AD:AB=3:4$。}
    \solstep{模型二：补出斜边}{在直角 $\triangle BCD$ 中，设 $BD=3y$、$BC=4y$，由勾股定理得 $CD=5y$。}
    \solstep{模型二：读取余弦}{$\cos C=\dfrac{BC}{CD}=\dfrac45$。}
    
  \end{solutionblock}
\end{problem}

\begin{problem}[points=12]
  \noindent
  \begin{minipage}[t]{\dimexpr\linewidth-60mm-6mm\relax}
    \vspace{0pt}
    如图，在 $\triangle ABC$ 中，点 $D$ 在线段 $AC$ 上，$BD\perp BC$，且
$\angle ABD=\angle ACB$。若 $\cos C=\dfrac{\sqrt3}{2}$，求 $AD:CD$。

    
  \end{minipage}\hfill
  \begin{diagramcoltikz}{60mm}{第 4 题图}
  \input{\detokenize{build/diagram/jobs/practice-q4-prompt/rendered/prompt.fragment.tex}}
  \end{diagramcoltikz}
  \par\medskip
  \answerarea[50mm]
  \setcounter{solstep}{0}
  \begin{solutionblock}
    \textbf{答案：}$AD:CD=1:2$\par\medskip
    \solstep{模型二：由余弦还原三边比}{设 $CD=2x$，则 $BC=\sqrt3x$。由勾股定理得 $BD=x$，所以 $BD:BC=1:\sqrt3$。}
    \solstep{模型一：接入相似对应边}{$\because \triangle ABD\sim\triangle ACB$\\ $\therefore \dfrac{AD}{AB}=\dfrac{AB}{AC}=\dfrac{BD}{BC}=\dfrac1{\sqrt3}$}
    \solstep{模型一：还原共线分段比}{设 $AD=y$、$AB=\sqrt3y$，则由 $AB^2=AD\cdot AC$ 得 $AC=3y$，所以 $CD=2y$，$AD:CD=1:2$。}
    
  \end{solutionblock}
\end{problem}
