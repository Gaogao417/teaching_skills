\eduSectionTitle{先相似定尺度，再用两种方法求 $GF$}

\begin{eduProblemBlock}[第 12 题]
\noindent
\begin{minipage}[t]{\dimexpr\linewidth-70mm-6mm\relax}
\vspace{0pt}
如图，在正方形 $ABCD$ 中，$E$ 为 $AB$ 边的中点，$G,F$ 分别为 $AD,BC$ 边上的点。若 $AG=2$，$BF=4$，$\angle GEF=90^\circ$，求 $GF$ 的长。

\end{minipage}\hfill

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{公共入口}
        \eduSideHint{相似顺序怎么对}{先对直角：$A\leftrightarrow B$；再对相等锐角：$G\leftrightarrow E$；最后才是 $E\leftrightarrow F$。}
        \eduSideMistake{根为什么取正}{$x$ 表示线段长度，所以由 $x^2=8$ 只取 $x=2\sqrt2$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{先用相似确定边长}
\needspace{5\baselineskip}      \eduExplainStep{}{先证相似，求出 $AE,BE$}{$\because \angle GAE=\angle EBF=90^\circ$，且 $GE\perp EF$、$AE\perp BF$，\\
$\therefore \angle AGE=\angle BEF$。\\
$\therefore \triangle AGE\sim\triangle BEF$。\\
设 $AE=BE=x$，则 $\dfrac{AG}{BE}=\dfrac{AE}{BF}$，即 $\dfrac2x=\dfrac x4$。\\
解得 $x=2\sqrt2$，所以 $AE=BE=2\sqrt2$，$AB=4\sqrt2$。
}
  \end{eduExplainRight}
\end{eduDualExplain}









\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{方法一观察}
        \eduSideHint{在哪个三角形收尾}{题设已给 $GE\perp EF$，所以只要分别求出两条直角边 $GE,EF$，就能在 $\triangle GEF$ 中求 $GF$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{求法一：利用题设直角}
\needspace{5\baselineskip}      \eduExplainStep{}{求 $GE,EF$，再在 $\triangle GEF$ 中勾股}{\textbf{解直角 $\triangle AGE$}：已知 $\angle GAE=90^\circ$，$AG=2$，$AE=2\sqrt2$，得 $GE=2\sqrt3$。\\
\textbf{解直角 $\triangle EBF$}：已知 $\angle EBF=90^\circ$，$BE=2\sqrt2$，$BF=4$，得 $EF=2\sqrt6$。\\
$\because \angle GEF=90^\circ$，\\
$\therefore GF=\sqrt{GE^2+EF^2}=\sqrt{12+24}=6$。
}
  \end{eduExplainRight}
\end{eduDualExplain}









\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{方法二观察}
        \eduSideHint{辅助线解决什么}{给梯形 $ABFG$ 作高后，$ABHG$ 成为矩形，能直接读出 $GH=AB$、$BH=AG$。}
        \eduSideHint{在哪个三角形收尾}{方法二不再计算 $GE,EF$，而是在新出现的直角三角形 $GHF$ 中一次勾股。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{求法二：梯形添高}
\needspace{19\baselineskip}      \eduExplainStep{}{过 $G$ 作 $GH\perp BF$ 于 $H$}{过 $G$ 作 $GH\perp BF$ 于 $H$。\\
$\because AG\parallel BF$，且 $AB\perp AG$，\\
$\therefore$ 四边形 $ABHG$ 为矩形。\\
$\therefore GH=AB=4\sqrt2$，$BH=AG=2$，$HF=BF-BH=2$。
}
      \begin{center}
        \begin{diagramcoltikz}{70mm}{补出 $GH$ 后，梯形 $ABFG$ 中出现矩形 $ABHG$ 和直角三角形 $GHF$。}
        \input{\detokenize{build/diagram/jobs/explanation-problem12-solution-gh/rendered/solution.fragment.tex}}
        \end{diagramcoltikz}
      \end{center}
\needspace{5\baselineskip}      \eduExplainStep{}{在 $\triangle GHF$ 中勾股，求出 $GF$}{\textbf{解直角 $\triangle GHF$}：已知 $\angle GHF=90^\circ$，$GH=4\sqrt2$，$HF=2$，\\
得 $GF=\sqrt{GH^2+HF^2}=\sqrt{32+4}=6$。
}
  \end{eduExplainRight}
\end{eduDualExplain}
