\eduSectionTitle{比例辅助线：解两组 A/8 字，把目标边标成份数}

\begin{eduProblemBlock}[例题]
\noindent
\begin{minipage}[t]{\dimexpr\linewidth-70mm-6mm\relax}
\vspace{0pt}
如图，点 $D$ 在线段 $BC$ 上，点 $E$ 在线段 $AC$ 上，$AD$ 与 $BE$ 交于点 $P$。已知 $AE:EC=2:3$，$BD:DC=4:5$，求 $AP:PD$。
\end{minipage}\hfill
\begin{diagramcoltikz}{70mm}{}
\input{\detokenize{build/diagram/jobs/explanation-example-prompt/rendered/prompt.fragment.tex}}
\end{diagramcoltikz}
\end{eduProblemBlock}










\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{看图提醒}
        \eduSideHint{颜色怎么读}{蓝字只表示题目给出的比；红字表示本步由相似得到的对应边份数。}
        \eduSideMistake{第二组先补整段}{$BD:DC=4:5$，所以 $BD:BC=4:9$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{19\baselineskip}      \eduExplainStep{1}{作辅助线}{过 $C$ 作 $CF\parallel AD$，交直线 $BE$ 于 $F$。}
      \begin{center}
        \begin{diagramcoltikz}{70mm}{蓝字标出题目给出的两组比；过 $C$ 作 $CF\parallel AD$。}
        \input{\detokenize{build/diagram/jobs/explanation-example-helper/rendered/solution.fragment.tex}}
        \end{diagramcoltikz}
      \end{center}
\needspace{19\baselineskip}      \eduExplainStep{2}{解第一组 8 字}{$\because CF\parallel AP$。\\ $\therefore \triangle EAP\sim\triangle ECF$。\\ $AE:EC=2:3$，故 $AP:CF=2:3$。}
      \begin{center}
        \begin{diagramcoltikz}{70mm}{第二步解 8 字：由蓝色 $2:3$ 标出红色 $AP:CF=2:3$。}
        \input{\detokenize{build/diagram/jobs/explanation-example-model1/rendered/solution.fragment.tex}}
        \end{diagramcoltikz}
      \end{center}
\needspace{19\baselineskip}      \eduExplainStep{3}{解第二组 A 字}{$BD:BC=4:(4+5)=4:9$。\\ $\because CF\parallel DP$。\\ $\therefore \triangle BDP\sim\triangle BCF$。沿用 $CF=3$ 份，故只标 $PD=\frac{4}{3}$ 份。}
      \begin{center}
        \begin{diagramcoltikz}{70mm}{第三步解 A 字：第一组红字保持不变，只新增 $PD=4/3$ 份。}
        \input{\detokenize{build/diagram/jobs/explanation-example-model2/rendered/solution.fragment.tex}}
        \end{diagramcoltikz}
      \end{center}
\needspace{5\baselineskip}      \eduExplainStep{4}{比较两条边的份数}{图中已有 $AP=2$ 份、$PD=\frac{4}{3}$ 份。\\ $\therefore AP:PD=2:\frac{4}{3}=3:2$。}
  \end{eduExplainRight}
\end{eduDualExplain}
