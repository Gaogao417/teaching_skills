\begin{eduAuxItem}{思路导航}
公共阶段 a：先证相似，求出 $AE,BE$\end{eduAuxItem}





\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{公共入口}
        \eduSideHint{相似顺序怎么对}{先对直角：$A\leftrightarrow B$；再对相等锐角：$G\leftrightarrow E$；最后才是 $E\leftrightarrow F$。}
        \eduSideMistake{根为什么取正}{$x$ 表示线段长度，所以由 $x^2=8$ 只取 $x=2\sqrt2$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{公共阶段 a}
      \eduExplainStep{1}{公共阶段 a：先证相似，求出 $AE,BE$}{$\because \angle GAE=\angle EBF=90^\circ$，且 $GE\perp EF$、$AE\perp BF$，\\
$\therefore \angle AGE=\angle BEF$。\\
$\therefore \triangle AGE\sim\triangle BEF$。\\
设 $AE=BE=x$，则 $\dfrac{AG}{BE}=\dfrac{AE}{BF}$，即 $\dfrac2x=\dfrac x4$。\\
解得 $x=2\sqrt2$，所以 $AE=BE=2\sqrt2$，$AB=4\sqrt2$。
}
  \end{eduExplainRight}
\end{eduDualExplain}




\begin{eduAuxItem}{思路导航}
方法一 b1：求 $GE,EF$，再在 $\triangle GEF$ 中勾股\end{eduAuxItem}





\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{方法一观察}
        \eduSideHint{在哪个三角形收尾}{题设已给 $GE\perp EF$，所以只要分别求出两条直角边 $GE,EF$，就能在 $\triangle GEF$ 中求 $GF$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{方法一 b1：利用题设直角}
      \eduExplainStep{1}{方法一 b1：求 $GE,EF$，再在 $\triangle GEF$ 中勾股}{\textbf{解直角 $\triangle AGE$}：已知 $\angle GAE=90^\circ$，$AG=2$，$AE=2\sqrt2$，得 $GE=2\sqrt3$。\\
\textbf{解直角 $\triangle EBF$}：已知 $\angle EBF=90^\circ$，$BE=2\sqrt2$，$BF=4$，得 $EF=2\sqrt6$。\\
$\because \angle GEF=90^\circ$，\\
$\therefore GF=\sqrt{GE^2+EF^2}=\sqrt{12+24}=6$。
}
  \end{eduExplainRight}
\end{eduDualExplain}




\begin{eduAuxItem}{思路导航}
方法二 b2：过 $G$ 作 $GH\perp BF$ 于 $H$，解梯形 $ABFG$$\;\to\;$在 $\triangle GHF$ 中勾股，求出 $GF$\end{eduAuxItem}





\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{方法二观察}
        \eduSideHint{辅助线解决什么}{给梯形 $ABFG$ 作高后，$ABHG$ 成为矩形，能直接读出 $GH=AB$、$BH=AG$。}
        \eduSideHint{在哪个三角形收尾}{方法二不再计算 $GE,EF$，而是在新出现的直角三角形 $GHF$ 中一次勾股。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{方法二 b2：梯形添高}
      \eduExplainStep{1}{方法二 b2：过 $G$ 作 $GH\perp BF$ 于 $H$，解梯形 $ABFG$}{过 $G$ 作 $GH\perp BF$ 于 $H$。\\
$\because AG\parallel BF$，且 $AB\perp AG$，\\
$\therefore$ 四边形 $ABHG$ 为矩形。\\
$\therefore GH=AB=4\sqrt2$，$BH=AG=2$，$HF=BF-BH=2$。
}
      \begin{center}
        \begin{diagramcoltikz}{70mm}{补出 $GH$ 后，梯形 $ABFG$ 中出现矩形 $ABHG$ 和直角三角形 $GHF$。}
        \input{\detokenize{build/diagram/jobs/explanation-problem12-solution-gh/rendered/solution.fragment.tex}}
        \end{diagramcoltikz}
      \end{center}
      \eduExplainStep{2}{在 $\triangle GHF$ 中勾股，求出 $GF$}{\textbf{解直角 $\triangle GHF$}：已知 $\angle GHF=90^\circ$，$GH=4\sqrt2$，$HF=2$，\\
得 $GF=\sqrt{GH^2+HF^2}=\sqrt{32+4}=6$。
}
  \end{eduExplainRight}
\end{eduDualExplain}

\eduSectionTitle{第 13 题：标等角找相似，用“握手”数出 6 对}

\begin{eduProblemBlock}[第 13 题]
\noindent
\begin{minipage}[t]{\dimexpr\linewidth-70mm-6mm\relax}
\vspace{0pt}
如图，$AD,BE$ 是 $\triangle ABC$ 的两条高，$AD,BE$ 相交于点 $O$，问图中共有多少对相似三角形。

\end{minipage}\hfill
\begin{diagramcoltikz}{70mm}{把 $\angle DBO$ 与 $\angle EAO$ 标成同款弧角，再找同时含弧角和直角的三角形。}
\input{\detokenize{build/diagram/jobs/explanation-problem13-prompt/rendered/prompt.fragment.tex}}
\end{diagramcoltikz}
\end{eduProblemBlock}





\begin{eduAuxItem}{思路导航}
先由两条高标出一组等角$\;\to\;$再找含弧角的四个直角三角形$\;\to\;$最后用“每个都有 3 个伙伴”计数，再除以 2 去重\end{eduAuxItem}





\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{小贴士}
        \eduSideHint{为什么一个弧角能找到两个}{$B$ 点的同一个弧角同时属于 $\triangle BDO$ 与 $\triangle BEC$；$A$ 点的同一个弧角同时属于 $\triangle AEO$ 与 $\triangle ADC$。}
        \eduSideHint{找三角形的筛子}{不要看大小和朝向，只检查两件事：有没有同款弧角？有没有 $90^\circ$？}
        \eduSideMistake{为什么要除以 2}{“$A$ 与 $B$ 相似”和“$B$ 与 $A$ 相似”只是顺序相反，表示同一对；$4\times3$ 把每一对恰好数了两次。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
      \eduExplainStep{1}{先由两条高标出一组等角}{$\because BE\perp AC$，\\
$\therefore \angle DBO=90^\circ-\angle C$。\\
$\because AD\perp BC$，\\
$\therefore \angle EAO=90^\circ-\angle C$。\\
$\therefore \angle DBO=\angle EAO$，把它们标成同款弧角。
}
      \eduExplainStep{2}{再找含弧角的四个直角三角形}{含 $B$ 点弧角的直角三角形：$\triangle BDO,\triangle BEC$。\\
含 $A$ 点弧角的直角三角形：$\triangle AEO,\triangle ADC$。\\
四个三角形都含一个弧角和一个直角，\\
$\therefore \triangle BDO\sim\triangle AEO\sim\triangle ADC\sim\triangle BEC$。
}
      \eduExplainStep{3}{最后用“每个都有 3 个伙伴”计数，再除以 2 去重}{四个三角形中，每个三角形都和另外三个三角形相似，先得 $4\times3=12$ 次。\\
但“$A$ 与 $B$ 相似”和“$B$ 与 $A$ 相似”是同一对，每一对都被重复计算了 $2$ 次。\\
所以相似三角形共有 $\dfrac{4\times3}{2}=6$ 对。
}
  \end{eduExplainRight}
\end{eduDualExplain}
