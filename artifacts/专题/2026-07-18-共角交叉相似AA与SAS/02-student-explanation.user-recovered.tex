\eduSectionTitle{经典模型：SAS 先锁角，AA 先锁一对角}

\begin{eduProblemBlock}[模型例题]
\noindent
\begin{minipage}[t]{\dimexpr\linewidth-70mm-6mm\relax}
\vspace{0pt}
如图，$A,B,D$ 三点共线，$A,C,E$ 三点共线，点序分别为 $A-B-D$ 和 $A-C-E$，$BE$ 与 $CD$ 交于点 $O$。已知 $\triangle ABC\sim\triangle AED$。


\begin{enumerate}[label=(\arabic*)]
  \item 证明 $\triangle ACD\sim\triangle ABE$；
  \item 证明 $\triangle BOC\sim\triangle DOE$；
  \item 证明 $\triangle BOD\sim\triangle COE$。
\end{enumerate}

\end{minipage}\hfill

\end{diagramcoltikz}
\end{eduProblemBlock}









\begin{eduSubQuestionItem}{(1)}
证明 $\triangle ACD\sim\triangle ABE$。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{SAS 小贴士}
        \eduSideHint{先找哪个角}{$\triangle ACD$ 和 $\triangle ABE$ 都以 $A$ 为顶点；共线让 $\angle CAD$ 与 $\angle BAE$ 实际上是同一个角。}
        \eduSideHint{夹边是哪四条}{角 $A$ 两侧只有 $AC,AD$ 与 $AB,AE$，所以目标比必须围绕这四条边来写。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}      {$\because \triangle ABC\sim\triangle AED$，\\
$\therefore \dfrac{AB}{AE}=\dfrac{AC}{AD}$，即 $AB\cdot AD=AC\cdot AE$。\\
$\therefore \dfrac{AC}{AB}=\dfrac{AD}{AE}$。\\
$\because \angle CAD=\angle BAE$，\\
$\therefore \triangle ACD\sim\triangle ABE$（SAS）。
}
  \end{eduExplainRight}
\end{eduDualExplain}




\begin{eduSubQuestionItem}{(2)}
证明 $\triangle BOC\sim\triangle DOE$。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}      {$\because BE$ 与 $CD$ 交于点 $O$，\\
$\therefore \angle BOC=\angle DOE$。\\
$\because \triangle ACD\sim\triangle ABE$，$\triangle ABC\sim\triangle AED$，\\
$\therefore \angle ABE=\angle ACD$，$\angle ABC=\angle AED$。\\
又 $\because \angle CAD=\angle DAE$，\\
$\therefore \angle ACD+\angle ADC=\angle AED+\angle ADE$，\\
$\therefore \angle ACD-\angle AED=\angle ADE-\angle ADC$。\\
而 $\angle OBC=\angle ABE-\angle ABC$，$\angle ODE=\angle ADE-\angle ADC$，\\
$\therefore \angle OBC=\angle ODE$。\\
$\therefore \triangle BOC\sim\triangle DOE$（AA）。
}
  \end{eduExplainRight}
\end{eduDualExplain}




\begin{eduSubQuestionItem}{(3)}
证明 $\triangle BOD\sim\triangle COE$。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{SAS 小贴士}
        \eduSideHint{先锁定夹角}{$\angle BOD$ 与 $\angle COE$ 是对顶角。一旦准备走 SAS，接下来就只看这两个角各自两侧的边。}
        \eduSideHint{夹边比从哪里来}{上一问的相似先给出 $\dfrac{BO}{DO}=\dfrac{CO}{EO}$；交叉相乘后重组，才能变成围着对顶角的 $\dfrac{BO}{CO}=\dfrac{DO}{EO}$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}      {$\because \triangle BOC\sim\triangle DOE$，\\
$\therefore \dfrac{BO}{DO}=\dfrac{CO}{EO}$，即 $BO\cdot EO=CO\cdot DO$。\\
$\therefore \dfrac{BO}{CO}=\dfrac{DO}{EO}$。\\
$\because \angle BOD=\angle COE$，\\
$\therefore \triangle BOD\sim\triangle COE$（SAS）。
}
  \end{eduExplainRight}
\end{eduDualExplain}
