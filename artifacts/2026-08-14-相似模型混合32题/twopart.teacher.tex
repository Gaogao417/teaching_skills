    \vspace{0pt}
    如图，$\angle PAB=\angle PDC$。

已知 $PB=2\sqrt{2}$，$AB=4$，$PC=4$。判断还可以求出哪条边，并求出它的长度。
    
  \end{minipage}\hfill
  \begin{diagramcoltikz}{60mm}{}
  \input{\detokenize{../题库/2026-07-16-反A形相似/items/Q044/build/diagram/jobs/question_bank-reverse_a-v2-q044-prompt/rendered/prompt.fragment.tex}}
  \end{diagramcoltikz}
  \par\medskip
  \answerarea[54mm]
  \setcounter{solstep}{0}
  \begin{solutionblock}
    \textbf{答案：}$DC=4\sqrt{2}$。\par\medskip
    \solstep{整理已知边长比}{题中给出的两边均为整数或整系数根式，因此可直接写成 $PB:AB=2\sqrt{2}:4$。}
    \solstep{由等角判相似}{由题设等角和构型自带的另一组等角，得 $\triangle PAB\sim\triangle PDC$。}
    \solstep{按顶点确认对应边}{三组小三角形边与大三角形边依次为：$PA\leftrightarrow PD$，$PB\leftrightarrow PC$，$AB\leftrightarrow DC$。}
    \solstep{保持比例方向一致}{统一保持两组对应边的方向，可列 $\dfrac{PB}{PC}=\dfrac{AB}{DC}$，再代入 $PC=4$。}
    \solstep{求值并验算}{解得 $DC=4\sqrt{2}$；代回后两组对应边的比完全相等。}
    
  \end{solutionblock}
\end{problem}
