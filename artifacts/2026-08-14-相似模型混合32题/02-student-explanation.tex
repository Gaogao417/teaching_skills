\begin{eduSubQuestionItem}{① 反A形}
求证 $\triangle ADE\sim\triangle ACB$，并写出它推出了什么。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{写一写}
        \eduSideHint{这次推出了什么？}{用 $\text{AA}$ 证相似，推出的是\textbf{边成比例} $\dfrac{AD}{AC}=\dfrac{AE}{AB}=\dfrac{DE}{CB}$；其中 $\dfrac{AD}{AC}=\dfrac{AE}{AB}$ 正是②的 $\text{SAS}$ 要用的条件。}
        \eduSideHint{对应怎么记}{$\angle ADE=\angle ACB$ 锁定 $D\leftrightarrow C$，公共角锁定 $A\leftrightarrow A$，于是 $E\leftrightarrow B$；是\textbf{交叉对应}，不是顺排。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}      \eduExplainStep{}{找等角}{$\angle DAE=\angle CAB$（公共角 $A$），又已知 $\angle ADE=\angle ACB$，凑成两对角分别相等。}
\needspace{5\baselineskip}      \eduExplainStep{}{选判定}{两对角分别相等，用 $\text{AA}$。}
\needspace{5\baselineskip}      \eduExplainStep{}{写格式}{在 $\triangle ADE$ 和 $\triangle ACB$ 中，\\
$\because \angle DAE=\angle CAB$（公共角）\\
$\quad\angle ADE=\angle ACB$（已知）\\
$\therefore \triangle ADE\sim\triangle ACB$（两角分别相等）
}
  \end{eduExplainRight}
\end{eduDualExplain}

\begin{eduSubQuestionItem}{② 共角换位}
求证 $\triangle ADC\sim\triangle AEB$，并写出它推出了什么。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{写一写}
        \eduSideHint{边比从哪来}{①得 $\dfrac{AD}{AC}=\dfrac{AE}{AB}$，交叉相乘 $AD\cdot AB=AC\cdot AE$，再交换内项得 $\dfrac{AD}{AE}=\dfrac{AC}{AB}$。}
        \eduSideHint{这次推出了什么？}{用 $\text{SAS}$ 证相似，推出的是\textbf{角相等} $\angle ADC=\angle AEB$；它取补角后正是③要用的条件。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}      \eduExplainStep{}{找等角}{$\angle DAC=\angle EAB$（同一个公共角 $A$，换成 $\triangle ADC$ 和 $\triangle AEB$）。}
\needspace{5\baselineskip}      \eduExplainStep{}{选判定}{把①的边比交叉相乘：$AD\cdot AB=AE\cdot AC$，即 $\dfrac{AD}{AE}=\dfrac{AC}{AB}$，夹角都是 $\angle A$，用 $\text{SAS}$。}
\needspace{5\baselineskip}      \eduExplainStep{}{写格式}{在 $\triangle ADC$ 和 $\triangle AEB$ 中，\\
$\because \angle DAC=\angle EAB$（公共角）\\
$\quad\dfrac{AD}{AE}=\dfrac{AC}{AB}$（由①交叉相乘）\\
$\therefore \triangle ADC\sim\triangle AEB$（两边成比例且夹角相等）
}
  \end{eduExplainRight}
\end{eduDualExplain}

\begin{eduSubQuestionItem}{③ 蝶形}
求证 $\triangle ODB\sim\triangle OEC$，并写出它推出了什么。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{写一写}
        \eduSideHint{角从哪来}{$D$ 在 $AB$ 上，$\angle ODB$ 与 $\angle ADC$ 互补；$E$ 在 $AC$ 上，$\angle OEC$ 与 $\angle AEB$ 互补。由②的等角取补角相等。}
        \eduSideHint{这次推出了什么？}{用 $\text{AA}$ 证相似，推出的是\textbf{边成比例} $\dfrac{OD}{OE}=\dfrac{OB}{OC}=\dfrac{DB}{EC}$；其中 $\dfrac{OD}{OE}=\dfrac{OB}{OC}$ 正是④要用的条件。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}      \eduExplainStep{}{找等角}{$O$ 是 $BE$ 与 $CD$ 的交点，$\angle DOB=\angle EOC$（对顶角）。}
\needspace{5\baselineskip}      \eduExplainStep{}{选判定}{由②的 $\angle ADC=\angle AEB$ 取补角：$\angle ODB=180^\circ-\angle ADC$，$\angle OEC=180^\circ-\angle AEB$，等角的补角相等，凑成第二对角，用 $\text{AA}$。}
\needspace{5\baselineskip}      \eduExplainStep{}{写格式}{在 $\triangle ODB$ 和 $\triangle OEC$ 中，\\
$\because \angle DOB=\angle EOC$（对顶角）\\
$\quad\angle ODB=\angle OEC$（等角的补角相等）\\
$\therefore \triangle ODB\sim\triangle OEC$（两角分别相等）
}
  \end{eduExplainRight}
\end{eduDualExplain}

\begin{eduSubQuestionItem}{④ 8字型}
求证 $\triangle ODE\sim\triangle OBC$，并写出它推出了什么。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{写一写}
        \eduSideHint{边比从哪来}{③得 $\dfrac{OD}{OE}=\dfrac{OB}{OC}$，交叉相乘 $OD\cdot OC=OB\cdot OE$，再交换内项得 $\dfrac{OD}{OB}=\dfrac{OE}{OC}$。}
        \eduSideHint{这次推出了什么？}{用 $\text{SAS}$ 证相似，推出的是\textbf{角相等} $\angle ODE=\angle OBC$、$\angle OED=\angle OCB$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}      \eduExplainStep{}{找等角}{$O$ 是 $BE$ 与 $CD$ 的交点，$\angle DOE=\angle BOC$（对顶角）。}
\needspace{5\baselineskip}      \eduExplainStep{}{选判定}{把③的边比交叉相乘：$OD\cdot OC=OE\cdot OB$，即 $\dfrac{OD}{OB}=\dfrac{OE}{OC}$，夹角都是 $\angle O$，用 $\text{SAS}$。}
\needspace{5\baselineskip}      \eduExplainStep{}{写格式}{在 $\triangle ODE$ 和 $\triangle OBC$ 中，\\
$\because \angle DOE=\angle BOC$（对顶角）\\
$\quad\dfrac{OD}{OB}=\dfrac{OE}{OC}$（由③交叉相乘）\\
$\therefore \triangle ODE\sim\triangle OBC$（两边成比例且夹角相等）
}
  \end{eduExplainRight}
\end{eduDualExplain}




\begin{eduSummaryBlock}{\eduIconPin}{一图四相似速记}
\begin{eduBulletList}
  \item \textbf{公共角 $A$} $\;\to\;$ ① 反A形 $\triangle ADE\sim\triangle ACB$（AA）、② 共角换位 $\triangle ADC\sim\triangle AEB$（SAS）。  \item \textbf{对顶角 $O$} $\;\to\;$ ③ 蝶形 $\triangle ODB\sim\triangle OEC$（AA）、④ 8字型 $\triangle ODE\sim\triangle OBC$（SAS）。  \item \textbf{结论喂养链}：①推出的边比 $\to$ ②的 SAS 用；②推出的角 $\to$ ③的 AA 用；③推出的边比 $\to$ ④的 SAS 用。上一组推出的，正是下一组要用的。  \item \textbf{清点结论}：用 $\text{AA}$ 证 $\to$ 推出\textbf{边成比例}；用 $\text{SAS}$ 证 $\to$ 推出\textbf{角相等}。\end{eduBulletList}
\end{eduSummaryBlock}
