\eduSectionTitle{反A一图四相似——找等角、选判定、写格式}


\begin{eduSolutionBox}{相似证明三步法}
  \begingroup
  \setlength{\parskip}{2.5mm}
\textbf{第一步 找等角}——在两个三角形里先找到一对相等的角：公共顶点处的\textbf{公共角}，或交点 $O$ 处的\textbf{对顶角}。\par
\textbf{第二步 选判定}——能找到第二对等角就用 $\textbf{AA}$；夹这个角的两组对应边成比例就用 $\textbf{SAS}$。\par
\textbf{第三步 写格式}——按固定格式书写：``在 $\triangle XXX$ 和 $\triangle YYY$ 中''$\to$写条件（$\because$）$\to$下结论（$\therefore$）。\par
\textbf{关键原则：用什么判定证，就推出另一种结论。}\\
用 $\text{AA}$ 证（给两对角相等）$\to$ 推出\textbf{三组对应边成比例}；\\
用 $\text{SAS}$ 证（给一对角 $+$ 夹角两组边比）$\to$ 推出\textbf{其余对应角相等}。
\par
  \endgroup
\end{eduSolutionBox}

\begin{eduSubQuestionItem}{① 反A形}
求证 $\triangle ADE\sim\triangle ACB$，并写出它推出了什么。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{写一写}
        \eduSideHint{这次推出了什么？}{用 $\text{AA}$ 证 $\to$ 推出\textbf{边成比例} $\dfrac{AD}{AC}=\dfrac{AE}{AB}$，下一组马上要用。}
        \eduSideHint{``反''在哪里}{对应是交叉的：$D\leftrightarrow C$，$E\leftrightarrow B$，不是 $D$ 对 $B$、$E$ 对 $C$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}\textbf{找等角：}$\angle DAE=\angle CAB$（$A$ 是公共顶点，$D$ 在 $AB$ 上、$E$ 在 $AC$ 上）。\\
\textbf{选判定：}已知 $\angle ADE=\angle ACB$，凑出第二对等角，用 $\text{AA}$。\\
在 $\triangle ADE$ 和 $\triangle ACB$ 中，\\
$\because \angle DAE=\angle CAB$（公共角）\\
$\quad\angle ADE=\angle ACB$（已知）\\
$\therefore \triangle ADE\sim\triangle ACB$（两角分别相等）\\\needspace{5\baselineskip}$\therefore \angle AED=\angle ABC$（角相等）\\
$\quad\dfrac{AD}{AC}=\dfrac{AE}{AB}=\dfrac{DE}{CB}$（边成比例）\\
\textit{用 AA 证相似，推出的是\textbf{边成比例}——它正是 ② 要用的条件。}  \end{eduExplainRight}
\end{eduDualExplain}








\begin{eduSubQuestionItem}{② 共角换位}
求证 $\triangle ADC\sim\triangle AEB$，并写出它推出了什么。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{写一写}
        \eduSideHint{边比从哪来}{① 得 $\dfrac{AD}{AC}=\dfrac{AE}{AB}$，交叉相乘得 $AD\cdot AB=AC\cdot AE$，再交换内项得 $\dfrac{AD}{AE}=\dfrac{AC}{AB}$。}
        \eduSideHint{这次推出了什么？}{用 $\text{SAS}$ 证 $\to$ 推出\textbf{角相等} $\angle ADC=\angle AEB$，下一组马上要用。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}\textbf{找等角：}$\angle DAC=\angle EAB$（同一个公共角 $A$，三角形换成 $\triangle ADC$ 和 $\triangle AEB$）。\\
\textbf{选判定：}把 ① 的边比交叉相乘：$AD\cdot AB=AE\cdot AC$，即 $\dfrac{AD}{AE}=\dfrac{AC}{AB}$，用 $\text{SAS}$。\\
在 $\triangle ADC$ 和 $\triangle AEB$ 中，\\
$\because \angle DAC=\angle EAB$（公共角）\\
$\quad\dfrac{AD}{AE}=\dfrac{AC}{AB}$（由①）\\
$\therefore \triangle ADC\sim\triangle AEB$（两边成比例且夹角相等）\\\needspace{5\baselineskip}$\therefore \angle ADC=\angle AEB,\ \angle ACD=\angle ABE$（角相等）\\
$\quad\dfrac{DC}{EB}=\dfrac{AD}{AE}=\dfrac{AC}{AB}$（边成比例）\\
\textit{用 SAS 证相似，推出的是\textbf{角相等}——$\angle ADC=\angle AEB$ 正是 ③ 要用的条件。}  \end{eduExplainRight}
\end{eduDualExplain}








\begin{eduSubQuestionItem}{③ 蝶形}
求证 $\triangle ODB\sim\triangle OEC$，并写出它推出了什么。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{写一写}
        \eduSideHint{角从哪来}{$D$ 在 $AB$ 上，$\angle ODB$ 与 $\angle ADC$ 互补；$E$ 在 $AC$ 上，$\angle OEC$ 与 $\angle AEB$ 互补。由②等角的补角相等。}
        \eduSideHint{为什么叫蝶形}{$\triangle ODB$ 和 $\triangle OEC$ 像一对翅膀挂在交点 $O$ 两侧；对应仍是交叉的：$D\leftrightarrow E$，$B\leftrightarrow C$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}\textbf{找等角：}$O$ 是 $BE$ 与 $CD$ 的交点，$\angle DOB=\angle EOC$（对顶角）。\\
\textbf{选判定：}由 ② 的 $\angle ADC=\angle AEB$ 取补角：$\angle ODB=180^\circ-\angle ADC$，$\angle OEC=180^\circ-\angle AEB$，等角的补角相等，凑出第二对等角，用 $\text{AA}$。\\
在 $\triangle ODB$ 和 $\triangle OEC$ 中，\\
$\because \angle DOB=\angle EOC$（对顶角）\\
$\quad\angle ODB=\angle OEC$（等角的补角相等）\\
$\therefore \triangle ODB\sim\triangle OEC$（两角分别相等）\\\needspace{5\baselineskip}$\therefore \dfrac{OD}{OE}=\dfrac{OB}{OC}=\dfrac{DB}{EC}$（边成比例）\\
\textit{用 AA 证相似，推出的是\textbf{边成比例}——它正是 ④ 要用的条件。}  \end{eduExplainRight}
\end{eduDualExplain}








\begin{eduSubQuestionItem}{④ 8字型}
求证 $\triangle ODE\sim\triangle OBC$，并写出它推出了什么。\end{eduSubQuestionItem}

\begin{eduDualExplain}
  \begin{eduExplainLeft}
    \eduColumnTitle{\eduIconBulb}{写一写}
        \eduSideHint{边比从哪来}{③ 得 $\dfrac{OD}{OE}=\dfrac{OB}{OC}$，交叉相乘得 $OD\cdot OC=OB\cdot OE$，再交换内项得 $\dfrac{OD}{OB}=\dfrac{OE}{OC}$。}
        \eduSideHint{8 字怎么看}{$DE$ 一横在上、$BC$ 一横在下，$BE$ 与 $CD$ 交叉穿过 $O$——整个就是数字 $8$。对应：$D\leftrightarrow B$，$E\leftrightarrow C$。}
  \end{eduExplainLeft}

  \begin{eduExplainRight}
    \eduColumnTitle{\eduIconBook}{解答}
\needspace{5\baselineskip}\textbf{找等角：}$\angle DOE=\angle BOC$（对顶角）。\\
\textbf{选判定：}把 ③ 的边比交叉相乘：$OD\cdot OC=OE\cdot OB$，即 $\dfrac{OD}{OB}=\dfrac{OE}{OC}$，用 $\text{SAS}$。\\
在 $\triangle ODE$ 和 $\triangle OBC$ 中，\\
$\because \angle DOE=\angle BOC$（对顶角）\\
$\quad\dfrac{OD}{OB}=\dfrac{OE}{OC}$（由③）\\
$\therefore \triangle ODE\sim\triangle OBC$（两边成比例且夹角相等）\\\needspace{5\baselineskip}$\therefore \angle ODE=\angle OBC,\ \angle OED=\angle OCB$（角相等）\\
$\quad\dfrac{DE}{BC}=\dfrac{OD}{OB}=\dfrac{OE}{OC}$（边成比例）\\
\textit{④ 的相似比与 ① 相同——$DE$ 在 $BC$ 上方，两线交于 $O$，正是``八''字。}  \end{eduExplainRight}
\end{eduDualExplain}




\begin{eduSummaryBlock}{\eduIconPin}{一图四相似速记}
\begin{eduBulletList}
  \item \textbf{公共角 $A$}$\;\to\;$① 反A形 $\triangle ADE\sim\triangle ACB$（AA）、② 共角换位 $\triangle ADC\sim\triangle AEB$（SAS）。  \item \textbf{对顶角 $O$}$\;\to\;$③ 蝶形 $\triangle ODB\sim\triangle OEC$（AA）、④ 8 字型 $\triangle ODE\sim\triangle OBC$（SAS）。  \item \textbf{链条}：①推出边比 $\to$ ②的 SAS 用；②推出角 $\to$ ③的 AA 用；③推出边比 $\to$ ④的 SAS 用。上一组推出的，正是下一组要用的。  \item \textbf{写结论}：用 $\text{AA}$ 证 $\to$ 推出\textbf{边成比例}；用 $\text{SAS}$ 证 $\to$ 推出\textbf{角相等}。\end{eduBulletList}
\end{eduSummaryBlock}
